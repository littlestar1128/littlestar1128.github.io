// value[]: 输出数组
UIDATA_DOUBLE *UIDATA_GetValue_Coord(UIDATA_SHORT id, UIDATA_DOUBLE value[])
{
    // ... 变量定义 ...
    int chan = ChannelSelect(); // 获取当前操作通道

    // [1. 获取机械坐标]
    if (id == UIDATA_ID_CurMachineCrd && value != NULL)
    {
        for (i = 0; i < 6; i++) // 遍历6个轴
        {
            // [2. 双通道交叉控制逻辑] (难点)
            // 场景：通道1借用通道2的轴进行加工
            
            // 情况A：2轴系统
            if(SysAxisNum[chan]==2) 
            {
                // 默认读取本通道坐标，经过 AxisChange 映射 (逻辑轴->物理轴)
                value[i] = MachineCrd[chan][AxisChange[i]];
                
                // 检查：是否开启交叉控制 (gIsCrossCtl)
                if(gIsCrossCtl && i<5)
                {
                    // 检查：对方通道(1-chan)的该轴是否被“借出” (gCrossAxisNo > 0)
                    if(gCrossAxisNo[1-chan][i] > 0)
                    {
                        // 关键点：读取【对方通道】的机械坐标，赋值给【本通道】的显示变量
                        value[i] = MachineCrd[1-chan][AxisChange[i]];
                    }
                }
            }
            
            // [3. M26 旋转轴特殊处理]
            // 如果该轴处于 M26 模式 (gYSpdFlg == 1)，说明它变成了主轴/旋转轴
            if(gYSpdFlg[chan][i] == 1) 
            {
                 // 读取 M26 专用坐标数组 (通常存的是角度或转速相关值)
                 value[i] = MachineCrd_M26[chan][i];
            }
        }
    }

    // [4. 车床工艺特殊处理]
    #if SOFTTYPE == LATHE
        // 直径编程逻辑：
        // 如果是 X轴 (i==0) 且 半径编程开关关闭 (RadiusPrg!=1)
        if(i==0 && BitParabuf[chan][RadiusPrg] == 0 ...)

             // 界面显示值 = 底层脉冲值 * 2
             // 因为车床图纸通常标直径，而电机控制的是半径
             value[i] = 2 * MachineCrd[chan][i];
        }

        // 这些是复合加工中心的特殊工艺，坐标来源不同

        {
             value[i] = MachineCrd_GunChi[chan][2]; // 读取滚齿专用坐标
        }
        
        // 尾座轴 (TailStock)
        if(BitParabuf[chan][TailStockEnable] == 1 ...)
        {
             value[i] = G_tailStockAxisMachCoord[...]; // 读取尾座坐标
        }
    #endif
  //  ........
if (GAxisType[i] == 1) { // 轴类型=1 代表旋转轴
    if(ABS(360.0 - value[i]) < 0.0000001) value[i] = 0.0;

//更新获取坐标值
void Coord_Update(UI_HANDLE_COORD hCoord, double *coord, ...)
{
    // ...
    int AxisNum = SysShowAxisNum(); // 获取当前需要显示的轴数量

    // [1. 针对 2轴系统的简单处理]
    if(AxisNum == 2)
    {
        // 直接格式化 coord[0] 和 coord[1] 写入第0行和第1行
        sprintf(str, "%9.4lf", coord[0]);
        LISTVIEW_SetItemText(hCoord, 1, 0, str);
        // ...
    }
    else
    {
        // [2. 通用处理循环]
        for(i=0; i<AxisNum; i++)
        {
            // [关键步骤]：先去列表里读取这一行显示的“轴名称”是什么？
            // 这样无论用户怎么配置轴顺序(X-Y-Z 还是 Z-X-Y)，显示都不会错。
            LISTVIEW_GetItemText(hCoord, 0, i, str, 50); // 读取第0列文本(如 "X", "Z")

            // [3. 名称匹配索引]
            switch (str[0]) 
            {
                case 'X': case 'U': k = 0; break; // 如果是 X 或 U，取数据数组第0个
                case 'Y': case 'V': k = 1; break; // 如果是 Y 或 V，取数据数组第1个
                case 'Z': case 'W': k = 2; break; // ...
                case 'A': k = 3; break;
                case 'B': k = 5; break; // 注意：B轴索引跳到了5
                case 'C': case 'H': k = 4; break;
            }

            // [4. 物理轴重映射]
            // 如果是2轴系统，可能存在逻辑轴到物理轴的二次映射
            if(SysAxisNum[chan]==2) k = AxisChange_Pos[k];

            // [5. 格式化并写入]
            // %.4lf: 保留4位小数 (CNC标准精度 0.1um)
            sprintf(str, "%9.4lf", coord[k]);
            LISTVIEW_SetItemText(hCoord, 1, i, str); // 写入第1列(数值列)
        }
    }
}

// [函数：M代码处理逻辑]
// 参数 ax: 当前指令作用的轴索引 (由 J 代码指定)
void PlcMstDeal_MCode(int chan, int ax)
{
    // ...
    #if SOFTTYPE == LATHE // 车床版软件
        // [1. 速度限制计算]
        // 如果未设置 M26专用限速 (参数为0)
        if(NumParabuf[chan][M26MaxSpeed] == 0)
        {
            // 使用该轴通用的最大速度 / 60
            // maxf: 最大脉冲频率 (Hz)
            Axis[chan][ax].maxf = NumParabuf[chan][MaxSpeed]/60; 
        }
        else
        {
            // [关键点：这就是导致 250 限速的原因]
            // 如果 M26MaxSpeed 小于 通用MaxSpeed，则使用 M26MaxSpeed
            if(NumParabuf[chan][M26MaxSpeed] < NumParabuf[chan][MaxSpeed])
            {
                // 计算公式：参数值 / 60
                // 如果参数是 90000，则 maxf = 1500 Hz
                Axis[chan][ax].maxf = NumParabuf[chan][M26MaxSpeed]/60; 
            }
            // else ... (省略)
        }
    #endif

    // [2. 状态保存]
    // 检查是否已经在旋转模式，如果不是 (!= ON)
    if(gYSpdFlg[chan][ax] != ON)
    {
        // 备份当前位置：因为变成主轴后，坐标就不再是线性位置了
        // 当 M27 (取消旋转) 时，需要恢复这个坐标
        AbsoluteCrd_M26[chan][ax] = AbsoluteCrd[chan][ax];
        MachineCrd_M26[chan][ax]  = MachineCrd[chan][ax];
    }

    // [3. 激活旋转标志]
    // 这是一个全局开关！
    // 一旦置位，UpdateDialog 和 UIDATA 就会改变显示逻辑
    gYSpdFlg[chan][ax] = ON; 

    // 通知 PLC 流程结束
    MSTCode2PLC[chan][0] = HCodeStep_End;
}
